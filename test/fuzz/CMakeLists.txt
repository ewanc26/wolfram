# libFuzzer harnesses for the untrusted-input decoders. Added only when
# WOLFRAM_BUILD_FUZZ is ON; the top-level CMakeLists.txt has already put the
# sanitizer flags on every target, dependencies included. docs/fuzzing.md
# describes the harnesses, the corpus and how to run them.

enable_testing()

set(WOLFRAM_FUZZ_HARNESSES
    fuzz_cbor
    fuzz_car
    fuzz_mst_node
    fuzz_commit
    fuzz_firehose_frame
    fuzz_label_frame
    fuzz_label_message
    fuzz_chat_mod_frame
    fuzz_chat_mod_event
    fuzz_jetstream
    fuzz_lex_decode)

# fuzz_lex_decode drives every generated *_decode_json function. The list is
# read from the generated header at configure time, so a regenerated
# atproto_lex.h is picked up without editing a table by hand.
set(lex_header ${PROJECT_SOURCE_DIR}/include/wolfram/atproto_lex.h)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${lex_header})
file(STRINGS ${lex_header} lex_lines REGEX "wf_lex_[a-z0-9_]+_decode_json\\(")
set(lex_table "/* Generated by test/fuzz/CMakeLists.txt from atproto_lex.h. */\n")
set(lex_count 0)
foreach(line IN LISTS lex_lines)
  string(REGEX MATCH "wf_lex_([a-z0-9_]+)_decode_json\\(" _ "${line}")
  string(APPEND lex_table "WF_FUZZ_LEX(${CMAKE_MATCH_1})\n")
  math(EXPR lex_count "${lex_count} + 1")
endforeach()
if(lex_count EQUAL 0)
  message(FATAL_ERROR "fuzz_lex_decode: no *_decode_json functions in ${lex_header}")
endif()
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/fuzz_lex_table.inc
     CONTENT "${lex_table}")

foreach(name IN LISTS WOLFRAM_FUZZ_HARNESSES)
  if(WOLFRAM_FUZZ_ENGINE STREQUAL "libfuzzer")
    add_executable(${name} ${name}.c)
    target_link_options(${name} PRIVATE -fsanitize=fuzzer)
  else()
    add_executable(${name} ${name}.c replay_main.c)
  endif()
  target_link_libraries(${name} PRIVATE wolfram)
  # Public headers such as agent.h include cJSON.h, as the tests' do.
  target_include_directories(
    ${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}
                    ${cjson_SOURCE_DIR} ${cjson_BINARY_DIR})
  # Replay the committed corpus once. libFuzzer with -runs=0 executes every
  # input and exits; the replay driver takes the same directory argument.
  set(corpus ${CMAKE_CURRENT_SOURCE_DIR}/corpus/${name})
  if(WOLFRAM_FUZZ_ENGINE STREQUAL "libfuzzer")
    add_test(NAME ${name}_replay COMMAND ${name} -runs=0 ${corpus})
  else()
    add_test(NAME ${name}_replay COMMAND ${name} ${corpus})
  endif()
  set_tests_properties(${name}_replay PROPERTIES LABELS fuzz)
endforeach()
